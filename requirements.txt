# ── Audio / STT / Speech ────────────────────────────────────
# gTTS + pygame are the DEFAULT speech path (tts_engine="gtts" in
# settings.json), so they are required, not optional. When they were listed as
# optional the device generated audio and silently discarded it — no sound and
# no error. tts.py now falls back to pyttsx3 and to a system player
# (afplay/aplay/mpg123), but install these for the intended voice quality.
pyttsx3>=2.90
gTTS>=2.5.0
pygame>=2.6.0
SpeechRecognition>=3.10.0
python-dotenv>=1.0.0

# vosk is the ONLY offline speech-recognition engine that actually works well
# here, and it needs a model as well as the package. Installing the package
# alone leaves voice input dependent on the internet, which on this device
# means Modes 3/4/7/9 stop working whenever the network does. After
# `pip install vosk`, download a model:
#
#   mkdir -p models_local && cd models_local
#   wget https://alphacephei.com/vosk/models/vosk-model-small-en-in-0.4.zip
#   unzip vosk-model-small-en-in-0.4.zip && mv vosk-model-small-en-in-0.4 vosk
#
# (~40MB download, 55MB on disk, Indian English.) It is deliberately placed on
# the SD card rather than under models/ — that path is a symlink to a
# removable USB stick, and speech is the device's primary input.
# Run `python3 selftest.py` afterwards to confirm it was found.
vosk>=0.3.45

# soundfile decodes gTTS MP3 in-memory so audio can be routed to a specific
# ALSA card. Without it (and without mpg123), Confidential Mode's
# private-earphones / public-speaker split cannot be enforced for the default
# voice — private text would play out of whichever card ALSA defaults to.
soundfile>=0.12.1

# Optional last-resort offline engine. Lower accuracy (~70-80%) than vosk and
# not required if vosk is installed.
pocketsphinx>=5.0.0

# ── Hardware I/O ────────────────────────────────────────────

# ── Optional (install if available on your platform) ────────
# These may require system-level libs or may not have wheels
# for every Python version. Install manually if pip fails.
# Everything below degrades gracefully when absent:
#
#   pip install PyAudio>=0.2.14          # VAD capture (hands-free voice)
#   pip install webrtcvad-wheels         # VAD capture (hands-free voice)
#   pip install sounddevice>=0.4.7
#   pip install soundfile>=0.12.1
#   pip install "mediapipe>=0.10.14,<0.10.35"   # Mode 5 gestures; no Py3.13 wheels yet
#   pip install librosa>=0.10.1          # emotion engine
#   pip install scikit-learn>=1.5.0      # emotion engine
#   pip install llama-cpp-python>=0.2.90 # offline LLM fallback
#   pip install surya-ocr                # high-accuracy OCR (slow on a Pi)
#   pip install argostranslate           # optional extra translation fallback
#
# System packages (not pip):
#   Tesseract OCR — macOS: brew install tesseract
#                   Pi:    sudo apt install tesseract-ocr
#   Audio player  — Pi:    sudo apt install alsa-utils mpg123
#                          (tts.py's fallback player; macOS uses built-in afplay)
#   espeak-ng     — Pi:    sudo apt install espeak-ng
#                          Ships the Hindi and Gujarati voices. Without it a
#                          translation can be produced offline but not spoken.
#
# ── Offline translation (Mode 4 without internet) ───────────
# High-quality English/Hindi/Gujarati translation is optional because its ML
# runtime and model weights are large. Install requirements-translation.txt,
# then provision the three local IndicTrans2 bundles described in
# docs/TRANSLATION_SETUP.md. Argos and a trusted local LibreTranslate server
# remain optional fallbacks; the built-in phrasebook is only an emergency set
# of common exact phrases and is not a general translator.
